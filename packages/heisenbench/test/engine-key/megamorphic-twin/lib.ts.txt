export interface Item {
	id: number;
	price: number;
	qty: number;
}

function make_items(n: number, batch: number): Item[] {
	const out: Item[] = [];
	for (let i = 0; i < n; i++) {
		out.push({ id: i, price: i * 3, qty: (i % 4) + 1 });
	}
	return out;
}

function total(items: Item[]): number {
	let s = 0;
	for (let i = 0; i < items.length; i++) {
		s += items[i].price * items[i].qty;
	}
	return s;
}

function make_rows(n: number): Array<{ label: string }> {
	const out: Array<{ label: string }> = [];
	for (let i = 0; i < n; i++) {
		const k = i % 6; out.push(k === 0 ? { label: 'row' + i } : k === 1 ? { label: 'a' + i } : k === 2 ? { label: 'b' + i } : k === 3 ? { label: 'c' + i } : k === 4 ? { label: 'd' + i } : { label: 'e' + i });
	}
	return out;
}

function label_len(rows: Array<{ label: string }>): number {
	let n = 0;
	for (const r of rows) {
		n += r.label.length;
	}
	return n;
}

export function run(n: number): number {
	let s = 0;
	for (let batch = 0; batch < 1000; batch++) {
		s += total(make_items(n, batch));
	}
	return s + label_len(make_rows(n));
}
