import { bench } from 'HEISENBENCH_BENCH';
import { count, order, total } from './lib.ts';

const sorted = (): number[] => Array.from({ length: 40 }, (_, i) => i * 7);

bench('order', ({ xs }) => order(xs), { setup: () => ({ xs: sorted() }) });

// The pattern is the code's, not the input's: the search grows and bends the text only.
const PAT = 'aaaaaaaaaaaaaaab';

bench('count', ({ text }) => count(text, PAT), {
	setup: () => ({
		text: 'the quick brown fox jumps over the lazy dog and a cat naps in a sunny spot'
	})
});

bench('total', ({ xs }) => total(xs), { setup: () => ({ xs: sorted() }) });
