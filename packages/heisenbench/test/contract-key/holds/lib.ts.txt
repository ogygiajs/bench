export interface Row {
	id: number;
	tags: string[];
}

function score(row: Row): number {
	let s = 0;
	for (const t of row.tags) {
		s += t.length;
	}
	return s;
}

/** Each row's score. @heisenbench in the middle of a line is a mention, not a contract.
 * @heisenbench linear in rows
 */
export function rank(rows: Row[]): number[] {
	const out: number[] = [];
	for (const r of rows) {
		out.push(score(r));
	}
	return out;
}
