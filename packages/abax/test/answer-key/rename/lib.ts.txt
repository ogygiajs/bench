export interface Row {
	id: number;
	tags: string[];
}

function tag_total(row: Row): number {
	let s = 0;
	for (const t of row.tags) {
		s += t.length;
	}
	return s;
}

export function rank(rows: Row[]): number[] {
	const out: number[] = [];
	for (const r of rows) {
		out.push(tag_total(r));
	}
	return out;
}
