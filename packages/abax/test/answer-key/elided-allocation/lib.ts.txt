export interface Row {
	id: number;
	tags: string[];
}

function score(row: Row): number {
	let s = 0;
	for (const t of row.tags) {
		const w = { len: t.length };
		s += w.len;
	}
	return s;
}

export function rank(rows: Row[]): number[] {
	const out: number[] = [];
	for (const r of rows) {
		out.push(score(r));
	}
	return out;
}
