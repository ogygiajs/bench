export function order(xs: number[]): number[] {
	return xs.slice().sort();
}

export function count(text: string, pat: string): number {
	return text.split(pat).length - 1;
}

export function total(xs: number[]): number {
	let s = 0;
	for (let i = 0; i < xs.length; i++) {
		s += xs[i];
	}
	return s;
}
