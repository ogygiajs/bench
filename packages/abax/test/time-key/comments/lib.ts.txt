// The loop the cases share.
function spin(n: number): number {
	let s = 0;
	// a running sum, kept small
	for (let i = 0; i < n; i++) s = (s + i * 7) % 1013;
	return s;
}

/** The measured work. */
export function work(): number {
	return spin(20000);
}

/** Work that no variant changes. */
export function steady(): number {
	return spin(2000);
}
